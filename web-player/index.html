1. Static HTML page (simplest)

Copy the player HTML skeleton, add CSS & JS, initialize with one line.

<!-- In <head> -->
<link rel="stylesheet" href="css/player.css">

<!-- In <body>: copy the #mv-player block from this file (keep data-mv attributes) -->
<div id="mv-player" class="mv-player">
  <video class="mv-player__video" poster="poster.jpg" playsinline></video>
  <!-- ... controls markup as in index.html ... -->
</div>

<!-- Before </body> -->
<script src="js/player.js"></script>
<script>
  const player = new ModernVideoPlayer(document.querySelector('#mv-player'), {
    src: 'https://commondatastorage.googleapis.com/gtv-videos-bucket/sample/BigBuckBunny.mp4',
    poster: 'poster.jpg',
    tracks: [{ label: 'English', lang: 'en', src: 'captions-en.vtt', default: true }],
    chapters: [{ time: 0, title: 'Intro' }, { time: 42, title: 'Chapter 1' }],
    qualities: [
      { label: '1080p', src: 'video-1080.mp4' },
      { label: '720p', src: 'video-720.mp4', default: true }
    ]
  });
</script>

2. Laravel Blade

Blade component — pass video data from controller.

{{-- resources/views/components/video-player.blade.php --}}
@props(['src', 'poster' => null, 'tracks' => []])
<link rel="stylesheet" href="{{ asset('css/player.css') }}">
<div class="mv-player" data-mv-auto-init data-options='{{ json_encode(["src"=>$src,"poster"=>$poster,"tracks"=>$tracks]) }}'>
  <!-- include same inner HTML as demo (video + controls + menus) -->
  <video class="mv-player__video" poster="{{ $poster }}"></video>
  <!-- ... rest of controls ... -->
</div>
<script src="{{ asset('js/player.js') }}"></script>

{{-- Usage in a Blade view --}}
<x-video-player :src="asset('videos/demo.mp4')" :tracks="[['label'=>'English','lang'=>'en','src'=>asset('captions/en.vtt')]]" />

3. React (or any SPA)

Wrap the vanilla class in a useEffect.

import { useEffect, useRef } from 'react';
import './player.css';

export default function VideoPlayer({ src, poster, tracks, playlist }) {
  const ref = useRef(null);
  useEffect(() => {
    let player;
    import('./player.js').then(() => {
      player = new window.ModernVideoPlayer(ref.current, { src, poster, tracks, playlist });
    });
    return () => player?.destroy();
  }, [src]);
  return (
    <div ref={ref} className="mv-player">
      {/* copy inner HTML from index.html (video + controls) */}
      <video className="mv-player__video" />
    </div>
  );
}
